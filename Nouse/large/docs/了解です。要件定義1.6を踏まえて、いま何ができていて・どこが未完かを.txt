```html
<script>
  // GitHub直配信用のフォールバック（raw -> jsDelivr）
  function withFallback(urls, onload, onerror) {
    const img = new Image();
    let idx = 0;
    function tryLoad() {
      img.src = urls[idx];
      img.onload = () => onload(urls[idx]);
      img.onerror = () => {
        idx++;
        if (idx < urls.length) tryLoad();
        else onerror && onerror();
      };
    }
    tryLoad();
  }

  function applyLensImages(leftEl, rightEl, src) {
    // src が1枚なら左右同一表示。2枚あるなら配列で渡す
    const leftSrcs = Array.isArray(src) ? [src[0]] : [src];
    const rightSrcs = Array.isArray(src) ? [src[1] || src[0]] : [src];

    // GitHub raw と jsDelivr を自動フォールバック
    const gh = (s) => s;
    const cdn = (s) => s.replace('https://raw.githubusercontent.com/', 'https://cdn.jsdelivr.net/gh/').replace('/main/', '@main/');
    const tryUrls = (s) => [gh(s), cdn(s)];

    withFallback(tryUrls(leftSrcs[0]), (ok) => {
      leftEl.style.backgroundImage = `url("${ok}")`;
    }, () => { leftEl.className = 'lens-image lens-pattern-1'; }); // 失敗時は既存パターンへ

    withFallback(tryUrls(rightSrcs[0]), (ok) => {
      rightEl.style.backgroundImage = `url("${ok}")`;
    }, () => { rightEl.className = 'lens-image lens-pattern-1'; });
  }
</script>
```

```js
// gas/DataAccess.gs
function getMasterSheet_() {
  return SpreadsheetApp.openById(CONFIG.MASTER_BOOK_ID).getSheetByName(CONFIG.MASTER_SHEET_NAME);
}

function pickRandom_(arr, n) {
  const a = arr.slice(); a.sort(() => Math.random() - 0.5);
  return a.slice(0, n);
}

function buildItem_(row) {
  // 必要に応じて列を調整
  var brand = row[CONFIG.BRAND_COL - 1];
  var color = row[CONFIG.COLOR_COL - 1];
  var imageUrl = row[CONFIG.IMAGE_URL_COL - 1]; // X列: フルURL推奨

  return {
    brand: brand,
    color: color,
    // 左右2枚あれば "URL1|URL2" などの書式も許容
    images: imageUrl && imageUrl.indexOf('|') > -1
      ? imageUrl.split('|').map(function(s){return s.trim();})
      : (imageUrl ? [imageUrl] : []),
    // 任意: ヒント・スペックも列から取り出して拡張可
  };
}

function getQuizItems_(limit) {
  var sh = getMasterSheet_();
  var last = sh.getLastRow();
  if (last < 2) return [];

  var values = sh.getRange(2, 1, last-2+1, sh.getLastColumn()).getValues();
  var items = values.map(buildItem_).filter(function(it){ return it.brand && it.color; });
  return pickRandom_(items, limit || 10);
}

// --- Web API ---
function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || 'items';
  var output;

  if (action === 'items') {
    output = { items: getQuizItems_(10) };
  } else if (action === 'ranking') {
    var scope = (e.parameter && e.parameter.scope) || 'daily';
    output = { ranking: getRanking_(scope) };
  } else if (action === 'mystats') {
    var user = (e.parameter && e.parameter.user) || '';
    output = getMyStats_(user);
  } else {
    output = { message: 'unknown action' };
  }

  return ContentService.createTextOutput(JSON.stringify(output))
    .setMimeType(ContentService.MimeType.JSON);
}
```

## B-3) `gas/AuthRank.gs`（成績保存＋ランキング算出）

```js
// gas/AuthRank.gs
function historySheet_() {
  return SpreadsheetApp.openById(CONFIG.QUIZ_HISTORY_BOOK_ID).getSheetByName(CONFIG.HISTORY_SHEET_NAME);
}

// 成績保存: POST /?action=submit
function doPost(e) {
  var params = JSON.parse(e.postData.contents || '{}');
  if (params.action === 'submit') {
    saveResult_(params);
    return ContentService.createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'unknown' }))
    .setMimeType(ContentService.MimeType.JSON);
}

function saveResult_(r) {
  // r = { user, store, score, correct, total, accuracy, timeSec, hints, dateISO }
  var sh = historySheet_();
  sh.appendRow([
    new Date(r.dateISO || new Date().toISOString()),
    r.user || '',
    r.store || '',
    r.score || 0,
    r.correct || 0,
    r.total || 10,
    r.accuracy || 0,
    r.timeSec || 0,
    r.hints || 0
  ]);
}

// ランキング集計
function getRanking_(scope) {
  var sh = historySheet_();
  var vals = sh.getDataRange().getValues();
  // ヘッダ仮定: [timestamp, user, store, score, correct, total, accuracy, timeSec, hints]
  var now = new Date();
  var from = new Date(now);
  if (scope === 'daily') from.setHours(0,0,0,0);
  if (scope === 'weekly') from.setDate(now.getDate() - 7);
  if (scope === 'monthly') from.setMonth(now.getMonth() - 1);

  var map = {};
  for (var i = 1; i < vals.length; i++) {
    var row = vals[i]; var ts = new Date(row[0]);
    if (ts < from) continue;
    var user = row[1] || 'unknown';
    var score = Number(row[3] || 0);
    var acc = Number(row[6] || 0);

    if (!map[user]) map[user] = { user: user, points: 0, counts: 0, accSum: 0, store: row[2] || '' };
    map[user].points += score;
    map[user].counts += 1;
    map[user].accSum += acc;
  }

  var arr = Object.keys(map).map(function(k){
    var o = map[k]; o.accRate = o.counts ? Math.round(o.accSum / o.counts) : 0; return o;
  });
  arr.sort(function(a,b){ return b.points - a.points; });
  return arr.slice(0, 50);
}

// マイ成績
function getMyStats_(user) {
  if (!user) return { error: 'no user' };
  var sh = historySheet_();
  var vals = sh.getDataRange().getValues();
  var recents = [];
  var all = { total:0, correct:0, played:0, points:0, timeSum:0, streak:0 };

  // 直近10回 / 通算
  for (var i=1; i<vals.length; i++){
    var row = vals[i];
    if (row[1] !== user) continue;
    var accuracy = Number(row[6] || 0);
    recents.push({ date: row[0], points: row[3], accuracy: accuracy });
    all.total += Number(row[5]||0);
    all.correct += Number(row[4]||0);
    all.points += Number(row[3]||0);
    all.timeSum += Number(row[7]||0);
    all.played += 1;
  }
  recents = recents.slice(-10);
  var avgTime = all.played ? Math.round((all.timeSum / all.played) * 10)/10 : 0;
  var accRate = all.total ? Math.round((all.correct / all.total) * 100) : 0;
  return { recents: recents, summary: { played: all.played, points: all.points, accuracy: accRate, avgTime: avgTime } };
}
```

### 3) JS（取得＆描画）

**外部ライブラリ不要版**（軽量）。Canvasに簡易棒/折れ線を描きます（Noto Sans JPのまま）。

```html
<script>
  function showStats() {
    showPage('statsPage');
    loadMyStats();
  }

  async function loadMyStats() {
    const user = encodeURIComponent(currentUser.name);
    const res = await fetch(`${GAS_ENDPOINT}?action=mystats&user=${user}`);
    const data = await res.json();

    // サマリ
    const sm = data.summary || {};
    document.getElementById('msPoints').textContent  = (sm.points ?? 0) + 'pt';
    document.getElementById('msAcc').textContent     = (sm.accuracy ?? 0) + '%';
    document.getElementById('msAvgTime').textContent = (sm.avgTime ?? 0) + '秒';
    document.getElementById('msPlayed').textContent  = (sm.played ?? 0) + '回';

    // 直近10回の簡易チャート（棒=points / 折れ線=accuracy）
    const cvs = document.getElementById('recentChart');
    const ctx = cvs.getContext('2d');
    ctx.clearRect(0,0,cvs.width,cvs.height);

    const rec = data.recents || [];
    if (!rec.length) { ctx.font = '14px "Noto Sans JP"'; ctx.fillText('データがありません', 10, 20); return; }

    // 正規化
    const W = cvs.width, H = cvs.height, P = 24;
    const maxPt = Math.max(1, ...rec.map(r=>Number(r.points||0)));
    // 軸
    ctx.strokeStyle='#e0e0e0'; ctx.beginPath(); ctx.moveTo(P, H-P); ctx.lineTo(W-P, H-P); ctx.stroke();

    // 棒（ポイント）
    const bw = (W - P*2) / rec.length * 0.6;
    rec.forEach((r,i)=>{
      const x = P + i * ((W - P*2)/rec.length) + (((W - P*2)/rec.length)-bw)/2;
      const h = Math.round((Number(r.points||0)/maxPt) * (H - P*2));
      ctx.fillStyle = '#FF6EC7';
      ctx.fillRect(x, H-P-h, bw, h);
    });

    // 折れ線（正答率）
    ctx.strokeStyle='#764BA2'; ctx.beginPath();
    rec.forEach((r,i)=>{
      const x = P + i * ((W - P*2)/rec.length) + (((W - P*2)/rec.length)/2);
      const y = H-P - Math.round((Number(r.accuracy||0)/100) * (H - P*2));
      if (i===0) ctx.moveTo(x,y); else ctx.lineTo(x,y);
      // マーカー
      ctx.fillStyle='#764BA2'; ctx.beginPath(); ctx.arc(x,y,2,0,Math.PI*2); ctx.fill();
    });
    ctx.stroke();

    // ラベル
    ctx.font = '12px "Noto Sans JP"'; ctx.fillStyle = '#666';
    ctx.fillText('ポイント', 10, 12);
    ctx.fillText('正答率', 70, 12);
  }
</script>
```

3. フロント `GAS_ENDPOINT` を差し替え、**データが手置き配列→APIのitemsに置換**されることを確認。
4. 10問終了で `submitResultToServer()` が呼ばれ、`quiz_history` に追記される。
5. ランキング：**タブ切替（デイリー/ウィークリー/マンスリー）** がそれぞれ値を更新する。
6. マイ成績：直近10回がグラフに出る／サマリの値（累計ポイント・平均正答率など）が表示される。
7. **Noto Sans JP** が全画面で効いている（グラフ描画の文字化けなし）。

## 補足（要件整合）

* **画像URLはX列(24)** を採用（GASの `buildItem_` 参照）。GitHub画像群の命名規則（品番/ブランド/カラー/装用期間）に合わせれば、フロント直リンク/差分更新も容易です。
* **GitHub直配信でOK**（社内利用・100GB/月枠内）。今回のフォールバックは raw→jsDelivr のみでCDN契約不要。
* 既存デモHTMLの構成（ページID、ランキングUI骨格、ナビ）を尊重し、**最小変更**で機能拡張しています。
